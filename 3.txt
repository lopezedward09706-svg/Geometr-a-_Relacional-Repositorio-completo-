def silencio_armonico(n_particles):
    """
    Genera el estado inicial |0⟩: un "océano" de fases en cancelación perfecta.
    Cada nodo tiene una fase aleatoria, pero el promedio es cero.
    """
    phases = np.random.uniform(-np.pi, np.pi, n_particles)
    amplitudes = np.ones(n_particles)  # Todas las ondas tienen amplitud unitaria
    # Superposición coherente: la suma vectorial tiende a cero
    real = amplitudes * np.cos(phases)
    imag = amplitudes * np.sin(phases)
    # Normalización para que el promedio sea cero (Silencio perfecto)
    real -= np.mean(real)
    imag -= np.mean(imag)
    return real + 1j * imag

def autocomparacion(state):
    """
    Aplica el operador de autocomparación: ⟨0|P̂|0⟩ = cos(0) = 1.
    Esto introduce una perturbación que rompe el Silencio.
    """
    # El operador de perturbación P̂ = cos(θ) con θ → 0
    perturbation = np.cos(0.0)  # = 1
    # La autocomparación proyecta el estado sobre sí mismo
    # Genera una "distinción" que se bifurca en A y B
    A = 1.0
    B = 1.0
    T = A + B  # Tensión total = 2
    # Rebotes fraccionarios (Fresnel)
    a = A / T
    b = B / T
    # Huella del tiempo (c)
    c = T - (a + b)  # = 1
    # Mediador expandido (C)
    C = A / a  # = 2
    # Deuda de Información
    D = A * B * C - 2 * a * b * c  # = 1.5
    return {
        'A': A, 'B': B, 'a': a, 'b': b, 'c': c, 'C': C,
        'T': T, 'D': D,
        'perturbation': perturbation
    }

def evolucion_fase(state, steps, dt):
    """
    Evoluciona el sistema de fases bajo la ecuación DNLS modificada.
    Simula la relajación del Silencio tras la perturbación.
    """
    history = [state.copy()]
    for step in range(steps):
        # Término de interacción lineal (acoplamiento elástico)
        interaccion = LAMBDA * np.roll(state, 1) + LAMBDA * np.roll(state, -1)
        # Término no lineal (auto-atrapamiento)
        no_lineal = 0.5 * np.abs(state)**2 * state
        # Disipación (fricción topológica) - solo para nodos periféricos
        disipacion = -1j * ETA * state * (np.abs(state) > 0.1)
        # Ecuación de evolución
        dstate = -1j * (interaccion + no_lineal) + disipacion
        state = state + dt * dstate
        # Normalización para conservar la norma
        state = state / np.sqrt(np.mean(np.abs(state)**2))
        history.append(state.copy())
    return history

def calcular_estadisticas(history):
    """
    Extrae métricas clave de la evolución.
    """
    amplitudes = np.abs(history)
    fases = np.angle(history)
    energia = np.mean(amplitudes**2, axis=1)
    entropia = -np.sum(amplitudes * np.log(amplitudes + 1e-12), axis=1)
    return {
        'amplitudes': amplitudes,
        'fases': fases,
        'energia': energia,
        'entropia': entropia
    }

# 1. GENERACIÓN DEL SILENCIO ARMÓNICO
# ------------------------------------------------------------

def generar_silencio_armonico(n_particles=None, seed=None):
    """
    Genera el estado inicial |0⟩: un "océano" de fases en cancelación perfecta.
    Cada nodo tiene una fase aleatoria, pero el promedio es cero.
    
    Parámetros:
    -----------
    n_particles : int, opcional
        Número de nodos de fase. Por defecto usa CONST['N_PARTICLES'].
    seed : int, opcional
        Semilla para reproducibilidad. Por defecto usa CONST['SEED'].
    
    Retorna:
    --------
    state : ndarray complex
        Vector de estado complejo que representa el Silencio Armónico.
    """
    if n_particles is None:
        n_particles = CONST['N_PARTICLES']
    if seed is not None:
        np.random.seed(seed)
    
    # Fases aleatorias uniformemente distribuidas
    phases = np.random.uniform(-np.pi, np.pi, n_particles)
    
    # Amplitudes unitarias (todas las ondas tienen la misma intensidad)
    amplitudes = np.ones(n_particles)
    
    # Superposición coherente: la suma vectorial tiende a cero
    real = amplitudes * np.cos(phases)
    imag = amplitudes * np.sin(phases)
    
    # Normalización para que el promedio sea cero (Silencio perfecto)
    real -= np.mean(real)
    imag -= np.mean(imag)
    
    return real + 1j * imag

# 2. AUTOCOMPARACIÓN — RUPTURA DEL SILENCIO
# ------------------------------------------------------------

def autocomparacion(state):
    """
    Aplica el operador de autocomparación: ⟨0|P̂|0⟩ = cos(0) = 1.
    Esto introduce una perturbación que rompe el Silencio y genera:
        - Bifurcación en A y B (tensión total T = 2)
        - Rebotes fraccionarios a y b (Fresnel)
        - Huella del tiempo c
        - Mediador expandido C
        - Deuda de Información 𝔇
    
    Parámetros:
    -----------
    state : ndarray complex
        Estado del Silencio Armónico.
    
    Retorna:
    --------
    params : dict
        Diccionario con todos los parámetros emergentes de la ruptura.
    """
    # La autocomparación genera una perturbación unitaria
    # P̂(0) = cos(0) = 1
    perturbacion = 1.0
    
    # Bifurcación: el "1" se duplica en dos ondas simétricas
    A = 1.0
    B = 1.0
    T = A + B  # Tensión total = 2
    
    # Rebotes fraccionarios (ley de Fresnel en la red)
    a = A / T  # 0.5
    b = B / T  # 0.5
    
    # Huella del tiempo: energía residual confinada en el centro
    c = T - (a + b)  # = 1
    
    # Mediador expandido: envoltura macroscópica
    C = A / a  # = 2
    
    # Deuda de Información: desequilibrio irresoluble
    # 𝔇 = ABC - 2abc = 1.5
    D = A * B * C - 2 * a * b * c
    
    # Verificación de la Ecuación Maestra
    abc = 2 * a * b * c
    ABC = A * B * C
    ecuacion_maestra_cumple = np.isclose(ABC, 4 * abc, rtol=1e-6)
    
    return {
        'A': A,
        'B': B,
        'a': a,
        'b': b,
        'c': c,
        'C': C,
        'T': T,
        'D': D,
        'ABC': ABC,
        'abc': abc,
        'perturbacion': perturbacion,
        'ecuacion_maestra': ecuacion_maestra_cumple,
        'estado_perturbado': state * perturbacion
    }

# MÓDULO 3: EVOLUCIÓN DINÁMICA (DNLS MODIFICADA)
# Geometría Relacional (RG) — Ecuación de Schrödinger No Lineal Discreta
# =============================================================================
# Este módulo implementa la evolución temporal del sistema de fases bajo la
# DNLS modificada, que incluye acoplamiento elástico, no linealidad y
# disipación por fricción topológica.
# =============================================================================

import numpy as np
from rg_constants import init_constants, LAMBDA, ETA

CONST = init_constants()

# ------------------------------------------------------------
# 1. FUNCIÓN DE EVOLUCIÓN DNLS MODIFICADA
# ------------------------------------------------------------

def dnls_evolucion(state, dt=None, lambda_val=None, eta_val=None):
    """
    Calcula la derivada temporal del estado según la DNLS modificada.
    
    Ecuación:
    dψ_i/dt = -i( Σ J_ij ψ_j + U|ψ_i|²ψ_i ) - iηψ_i (para nodos periféricos)
    
    Parámetros:
    -----------
    state : ndarray complex
        Estado actual del sistema.
    dt : float, opcional
        Paso de tiempo. Por defecto usa CONST['DT'].
    lambda_val : float, opcional
        Acoplamiento elástico. Por defecto usa LAMBDA.
    eta_val : float, opcional
        Fricción topológica. Por defecto usa ETA.
    
    Retorna:
    --------
    dstate : ndarray complex
        Derivada temporal del estado.
    """
    if dt is None:
        dt = CONST['DT']
    if lambda_val is None:
        lambda_val = LAMBDA
    if eta_val is None:
        eta_val = ETA
    
    # Término de interacción lineal (acoplamiento elástico entre vecinos)
    # Usamos un laplaciano discreto en 1D (para simplicidad)
    # En 2D/3D se usaría el operador de adyacencia de la red hexagonal
    interaccion = lambda_val * np.roll(state, 1) + lambda_val * np.roll(state, -1)
    
    # Término no lineal (auto-atrapamiento masivo)
    # U = 1.0 (constante de no linealidad)
    no_lineal = 1.0 * np.abs(state)**2 * state
    
    # Término disipativo (fricción topológica)
    # Solo aplica a nodos periféricos (amplitud > umbral)
    umbral = 0.1
    disipacion = -1j * eta_val * state * (np.abs(state) > umbral)
    
    # Ecuación de evolución completa
    dstate = -1j * (interaccion + no_lineal) + disipacion
    
    return dstate

# ------------------------------------------------------------
# 2. INTEGRADOR RUNGE-KUTTA DE 4° ORDEN (RK4)
# ------------------------------------------------------------

def rk4_step(state, dt=None):
    """
    Un paso de integración usando el método Runge-Kutta de 4° orden.
    
    Parámetros:
    -----------
    state : ndarray complex
        Estado actual.
    dt : float, opcional
        Paso de tiempo.
    
    Retorna:
    --------
    new_state : ndarray complex
        Estado después de un paso de tiempo.
    """
    if dt is None:
        dt = CONST['DT']
    
    k1 = dt * dnls_evolucion(state)
    k2 = dt * dnls_evolucion(state + 0.5 * k1)
    k3 = dt * dnls_evolucion(state + 0.5 * k2)
    k4 = dt * dnls_evolucion(state + k3)
    
    new_state = state + (k1 + 2*k2 + 2*k3 + k4) / 6.0
    
    return new_state

# ------------------------------------------------------------
# 3. EVOLUCIÓN COMPLETA
# ------------------------------------------------------------

def evolucionar_sistema(estado_inicial, n_steps=None, dt=None):
    """
    Evoluciona el sistema durante n_steps pasos de tiempo.
    
    Parámetros:
    -----------
    estado_inicial : ndarray complex
        Estado inicial del sistema.
    n_steps : int, opcional
        Número de pasos. Por defecto usa CONST['N_STEPS'].
    dt : float, opcional
        Paso de tiempo.
    
    Retorna:
    --------
    history : list
        Lista de estados en cada paso de tiempo.
    stats : dict
        Estadísticas de la evolución (energía, entropía, norma).
    """
    if n_steps is None:
        n_steps = CONST['N_STEPS']
    if dt is None:
        dt = CONST['DT']
    
    # Inicializar historial
    history = [estado_inicial.copy()]
    
    # Métricas
    energias = []
    entropias = []
    normas = []
    
    # Estado actual
    state = estado_inicial.copy()
    
    for step in range(n_steps):
        # Un paso RK4
        state = rk4_step(state, dt)
        
        # Normalización para conservar la norma (unitaridad)
        norma = np.sqrt(np.mean(np.abs(state)**2))
        if norma > 0:
            state = state / norma
        
        # Guardar estado
        history.append(state.copy())
        
        # Calcular métricas
        amplitudes = np.abs(state)
        energias.append(np.mean(amplitudes**2))
        # Entropía de Shannon (evita log(0))
        entropias.append(-np.sum(amplitudes * np.log(amplitudes + 1e-12)))
        normas.append(norma)
    
    stats = {
        'energia': np.array(energias),
        'entropia': np.array(entropias),
        'norma': np.array(normas)
    }
    
    return history, stats

import numpy as np

def analizar_fases(history):
    """
    Analiza la evolución de la distribución de fases.
    
    Parámetros:
    -----------
    history : list
        Lista de estados en cada paso de tiempo.
    
    Retorna:
    --------
    resultados : dict
        Métricas de la distribución de fases.
    """
    # Fases iniciales y finales
    fases_iniciales = np.angle(history[0])
    fases_finales = np.angle(history[-1])
    
    # Momentos estadísticos
    media_inicial = np.mean(fases_iniciales)
    media_final = np.mean(fases_finales)
    std_inicial = np.std(fases_iniciales)
    std_final = np.std(fases_finales)
    
    # Picos en la distribución final (triada ABC)
    # Buscamos picos en 0, π/3, 2π/3
    hist, bins = np.histogram(fases_finales, bins=50)
    bin_centers = (bins[:-1] + bins[1:]) / 2
    
    # Detectar picos
    from scipy.signal import find_peaks
    peaks, _ = find_peaks(hist, height=np.max(hist) * 0.3)
    peak_positions = bin_centers[peaks]
    
    # Picos esperados de la triada ABC
    expected_peaks = [0, np.pi/3, 2*np.pi/3]
    found_peaks = []
    for ep in expected_peaks:
        # Buscar el pico más cercano
        if len(peak_positions) > 0:
            idx = np.argmin(np.abs(peak_positions - ep))
            if np.abs(peak_positions[idx] - ep) < 0.3:
                found_peaks.append(peak_positions[idx])
    
    return {
        'media_inicial': media_inicial,
        'media_final': media_final,
        'std_inicial': std_inicial,
        'std_final': std_final,
        'picos_encontrados': found_peaks,
        'picos_esperados': expected_peaks,
        'triada_detectada': len(found_peaks) >= 2
    }

def crear_animacion(history, intervalo=50):
    """
    Crea una animación de la evolución del sistema en el plano de fase.
    
    Parámetros:
    -----------
    history : list
        Lista de estados en cada paso de tiempo.
    intervalo : int
        Intervalo entre frames en milisegundos.
    
    Retorna:
    --------
    ani : matplotlib.animation.FuncAnimation
        Objeto de animación.
    """
    fig, ax = plt.subplots(figsize=(8, 8))
    ax.set_xlim(-2, 2)
    ax.set_ylim(-2, 2)
    ax.set_title('Evolución del Silencio → Ruptura', fontsize=14)
    ax.set_xlabel('Real')
    ax.set_ylabel('Imaginario')
    ax.grid(alpha=0.3)
    ax.set_aspect('equal')
    
    scatter = ax.scatter([], [], c=[], cmap='viridis', alpha=0.6, s=10)
    centro_circle = Circle((0,0), 0.1, color='red', alpha=0.5)
    ax.add_artist(centro_circle)
    
    def update(frame):
        state = history[frame]
        scatter.set_offsets(np.column_stack([np.real(state), np.imag(state)]))
        scatter.set_array(np.abs(state))
        ax.set_title(f'Evolución del Silencio → Ruptura (paso {frame}/{len(history)-1})')
        return scatter,
    
    ani = animation.FuncAnimation(fig, update, frames=len(history),
                                  interval=intervalo, blit=True)
    return ani
