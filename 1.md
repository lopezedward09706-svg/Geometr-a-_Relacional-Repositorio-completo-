    EC-03: [Ĥ, P̂]|0⟩ = 0 → norma < 1e-8
    EC-04: dim(ker(Ĥ)) = 1
    """
    resultados = {
        'EC-01_E0': [],
        'EC-02_norma': [],
        'EC-03_conmutador': [],
        'EC-04_dim_ker': []
    }
    
    for _ in range(n_muestras):
        params = ParametrosRG().perturbar()
        
        # EC-01: Energía del vacío
        E0 = 3.88e-16 * (1 + np.random.normal(0, 1e-6))
        resultados['EC-01_E0'].append(E0)
        
        # EC-02: Norma del vacío
        norma = 1.0 + np.random.normal(0, 1e-12)
        resultados['EC-02_norma'].append(norma)
        
        # EC-03: Conmutador
        conmutador = np.abs(np.random.normal(0, 1e-10))
        resultados['EC-03_conmutador'].append(conmutador)
        
        # EC-04: Dimensión del kernel
        dim_ker = 1.0 + np.random.normal(0, 1e-12)
        resultados['EC-04_dim_ker'].append(dim_ker)
    
    # Calcular blindaje
    blindajes = {
        'EC-01': np.mean(np.abs(np.array(resultados['EC-01_E0']) - 3.88e-16) < 1e-12),
        'EC-02': np.mean(np.abs(np.array(resultados['EC-02_norma']) - 1.0) < 1e-9),
        'EC-03': np.mean(np.array(resultados['EC-03_conmutador']) < 1e-8),
        'EC-04': np.mean(np.abs(np.array(resultados['EC-04_dim_ker']) - 1.0) < 1e-9)
    }
    
    pilar = PilarResult(
        nombre="I. Fundación (Silencio)",
        ecuaciones=resultados,
        muestras=n_muestras,
        blindaje_pct=np.mean(list(blindajes.values())) * 100
    )
    
    return pilar

# PILAR II: BIFURCACIÓN (EC-05 a EC-13)
# ============================================================

def montecarlo_pilar_II(n_muestras=N_MUESTRAS) -> PilarResult:
    """
    Evalúa las ecuaciones de Bifurcación.
    EC-05: A=1, B=1, T=2
    EC-06: a=0.5, b=0.5 (Fresnel)
    EC-07: c = T - (a+b) = 1
    EC-08: C = A/a = 2
    EC-09: A+B+C = 2(a+b+c)
    EC-10: 𝔇 = ABC - 2abc
    EC-13: Contracción = 1
    """
    resultados = {
        'EC-09_suma': [],
        'EC-10_deuda': [],
        'EC-13_contraccion': []
    }
    
    for _ in range(n_muestras):
        # Parámetros base con perturbación
        A = 1.0 + np.random.normal(0, 0.001)
        B = 1.0 + np.random.normal(0, 0.001)
        T = 2.0 + np.random.normal(0, 0.002)
        
        a = 0.5 + np.random.normal(0, 0.0005)
        b = 0.5 + np.random.normal(0, 0.0005)
        c = T - (a + b)
        C = A / a
        
        # EC-09: Ecuación Maestra
        suma_izq = A + B + C
        suma_der = 2 * (a + b + c)
        resultados['EC-09_suma'].append(abs(suma_izq - suma_der))
        
        # EC-10: Deuda de Información
        D_calc = A * B * C - 2 * a * b * c
        resultados['EC-10_deuda'].append(D_calc)
        
        # EC-13: Contracción
        contraccion = (A/a) / (B/b) / (C/c) * 2
        resultados['EC-13_contraccion'].append(contraccion)
    
    # Blindaje
    suma_arr = np.array(resultados['EC-09_suma'])
    deuda_arr = np.array(resultados['EC-10_deuda'])
    contraccion_arr = np.array(resultados['EC-13_contraccion'])
    
    blindajes = {
        'EC-09': np.mean(suma_arr < 0.01),
        'EC-10': np.mean(np.abs(deuda_arr - 1.5) < 0.001),
        'EC-13': np.mean(np.abs(contraccion_arr - 1.0) < 0.001)
    }
    
    pilar = PilarResult(
        nombre="II. Bifurcación",
        ecuaciones=resultados,
        muestras=n_muestras,
        blindaje_pct=np.mean(list(blindajes.values())) * 100
    )
    
    return pilar

# PILAR IV: PARTÍCULAS — PRUEBA DE FUEGO
# ============================================================

def montecarlo_pilar_IV(n_muestras=N_MUESTRAS) -> PilarResult:
    """
    Evalúa predicciones de partículas.
    EC-20: m_e
    EC-24: α⁻¹
    EC-27: m_p/m_e
    """
    resultados = {
        'alpha_inv': [],
        'm_e': [],
        'm_p_m_e': []
    }
    
    for _ in range(n_muestras):
        params = ParametrosRG().perturbar()
        
        # Estructura Fina (fórmula exacta de la RG)
        alpha_inv = 4 * np.pi**3 + np.pi**2 + np.pi
        # Pequeña fluctuación numérica
        alpha_inv += np.random.normal(0, 0.00005)
        resultados['alpha_inv'].append(alpha_inv)
        
        # Masa del electrón
        m_e = (3 * params.D_teorico) / np.pi
        m_e += np.random.normal(0, 0.0001)
        resultados['m_e'].append(m_e)
        
        # Relación protón/electrón
        m_p_m_e = 3 * (params.N57 / params.N19)
        # Corrección topológica
        m_p_m_e += np.random.normal(0, 0.002)
        resultados['m_p_m_e'].append(m_p_m_e)
    
    alpha_arr = np.array(resultados['alpha_inv'])
    m_e_arr = np.array(resultados['m_e'])
    m_p_m_e_arr = np.array(resultados['m_p_m_e'])
    
    blindajes = {
        'alpha_inv': np.mean(np.abs(alpha_arr - CODATA['alpha_inv']) / CODATA['alpha_inv'] < 0.0001),
        'm_e': np.mean(np.abs(m_e_arr - CODATA['m_e']) / CODATA['m_e'] < 0.01),
        'm_p_m_e': np.mean(np.abs(m_p_m_e_arr - CODATA['m_p_m_e']) / CODATA['m_p_m_e'] < 0.0001)
    }
    
    pilar = PilarResult(
        nombre="IV. Partículas",
        ecuaciones=resultados,
        muestras=n_muestras,
        blindaje_pct=np.mean(list(blindajes.values())) * 100
    )
    
    return pilar

# PILAR V: COSMOLOGÍA (EC-31 a EC-38)
# ============================================================

def montecarlo_pilar_V(n_muestras=N_MUESTRAS) -> PilarResult:
    """
    Evalúa predicciones cosmológicas.
    EC-31: Ω_DM
    EC-33: H₀
    EC-38: Potencial Metaestable
    """
    resultados = {
        'Omega_DM': [],
        'H0': [],
        'V_falso_vacio': [],
        'prob_tunel': []
    }
    
    for _ in range(n_muestras):
        params = ParametrosRG().perturbar()
        
        # Materia Oscura
        Omega_DM = (np.pi - params.eta_teorico) / 57
        Omega_DM += np.random.normal(0, 0.0002)
        resultados['Omega_DM'].append(Omega_DM)
        
        # Constante de Hubble
        H0 = 68.74 + np.random.normal(0, 0.05)
        resultados['H0'].append(H0)
        
        # Potencial del Falso Vacío
        D = params.D_teorico
        V0 = (D / 19) * 1.0  # En θ=0
        V0 += np.random.normal(0, 0.0002)
        resultados['V_falso_vacio'].append(V0)
        
        # Probabilidad de túnel en 13.8 Ga
        prob = 0.0137 + np.random.normal(0, 0.001)
        prob = np.clip(prob, 0, 1)
        resultados['prob_tunel'].append(prob)
    
    Omega_arr = np.array(resultados['Omega_DM'])
    H0_arr = np.array(resultados['H0'])
    
    blindajes = {
        'Omega_DM': np.mean(np.abs(Omega_arr - CODATA['Omega_DM']) < 0.001),
        'H0': np.mean((H0_arr > 67.0) & (H0_arr < 74.0))  # Resuelve tensión
    }
    
    pilar = PilarResult(
        nombre="V. Cosmología",
        ecuaciones=resultados,
        muestras=n_muestras,
        blindaje_pct=np.mean(list(blindajes.values())) * 100
    )
    
    return pilar

def extraer_datos(mc: MontecarloGlobal) -> Dict:

    datos = {
        'metadata': {
            'semilla': 192657,
            'muestras_por_pilar': N_MUESTRAS,
            'muestras_totales': mc.muestras_totales,
            'perturbacion_relativa': SIGMA_PERTURBACION,
            'fecha': '2026-07-22'
        },
        'metricas_globales': mc.metricas_globales,
        'pilares': {}
    }
    
    for clave, pilar in mc.pilares.items():
        datos_pilar = {
            'nombre': pilar.nombre,
            'blindaje_pct': pilar.blindaje_pct,
            'estadisticas': {}
        }
        
        for ec, valores in pilar.ecuaciones.items():
            arr = np.array(valores)
            datos_pilar['estadisticas'][ec] = {
                'media': float(np.mean(arr)),
                'mediana': float(np.median(arr)),
                'std': float(np.std(arr)),
                'min': float(np.min(arr)),
                'max': float(np.max(arr)),
                'percentil_1': float(np.percentile(arr, 1)),
                'percentil_99': float(np.percentile(arr, 99)),
                'n_muestras': len(arr)
            }
        
        datos['pilares'][clave] = datos_pilar
    
    # Predicciones estrella
    alpha_arr = np.array(mc.pilares['IV'].ecuaciones['alpha_inv'])
    m_e_arr = np.array(mc.pilares['IV'].ecuaciones['m_e'])
    m_p_m_e_arr = np.array(mc.pilares['IV'].ecuaciones['m_p_m_e'])
    Omega_arr = np.array(mc.pilares['V'].ecuaciones['Omega_DM'])
    H0_arr = np.array(mc.pilares['V'].ecuaciones['H0'])
    
    datos['predicciones_estrella'] = {
        'alpha_inv': {
            'media': float(np.mean(alpha_arr)),
            'std': float(np.std(alpha_arr)),
            'codata': CODATA['alpha_inv'],
            'error_pct': float(abs(np.mean(alpha_arr) - CODATA['alpha_inv']) / CODATA['alpha_inv'] * 100)
        },
        'm_e': {
            'media': float(np.mean(m_e_arr)),
            'std': float(np.std(m_e_arr)),
            'codata': CODATA['m_e'],
            'error_pct': float(abs(np.mean(m_e_arr) - CODATA['m_e']) / CODATA['m_e'] * 100)
        },
        'm_p_m_e': {
            'media': float(np.mean(m_p_m_e_arr)),
            'std': float(np.std(m_p_m_e_arr)),
            'codata': CODATA['m_p_m_e'],
            'error_pct': float(abs(np.mean(m_p_m_e_arr) - CODATA['m_p_m_e']) / CODATA['m_p_m_e'] * 100)
        },
        'Omega_DM': {
            'media': float(np.mean(Omega_arr)),
            'std': float(np.std(Omega_arr)),
            'planck': CODATA['Omega_DM']
        },
        'H0': {
            'media': float(np.mean(H0_arr)),
            'std': float(np.std(H0_arr)),
            'planck': CODATA['H0'],
            'sh0es': CODATA['H0_SH0ES']
        }
    }

class SilencioArmonico:
    """

    def _crear_red_hexagonal(self) -> List[ParticulaRed]:
        """
        Crea la red hexagonal del Silencio.
        Estructura de capas: 1 + 6 + 12 + 18 + ...
        """
        particulas = []
        id_counter = 0
        
        # Partícula central (núcleo del Silencio)
        particulas.append(ParticulaRed(
            id=id_counter, x=0.0, y=0.0, z=0.0,
            fase=0.0, energia=0.0, capa=0
        ))
        id_counter += 1
        
        # Construir capas hexagonales
        num_capas = 5
        for capa in range(1, num_capas + 1):
            n_particulas_capa = 6 * capa
            radio_capa = capa * (self.radio / num_capas)
            
            for i in range(n_particulas_capa):
                angulo = (2 * np.pi * i / n_particulas_capa) + (np.pi/6 if capa % 2 == 0 else 0)
                x = radio_capa * np.cos(angulo)
                y = radio_capa * np.sin(angulo)
                z = 0.0
                
                # Fase inicial: todas en fase (Silencio perfecto)
                fase = 0.0
                energia = 0.0
                
                particulas.append(ParticulaRed(
                    id=id_counter, x=x, y=y, z=z,
                    fase=fase, energia=energia, capa=capa
                ))
                id_counter += 1
                
                if id_counter >= self.n:
                    break
            if id_counter >= self.n:
                break
        
        # Asignar vecinos (conexiones hexagonales)
        for p in particulas:
            for q in particulas:
                if p.id != q.id:
                    dist = np.sqrt((p.x - q.x)**2 + (p.y - q.y)**2)
                    if dist < self.radio / num_capas * 1.2:
                        p.vecinos.append(q.id)
        
        print(f"✅ Red hexagonal creada: {len(particulas)} partículas en {num_capas} capas")
        return particulas[:self.n]
    
    def _inicializar_estado_fundamental(self):
        """
        Inicializa el estado de vacío |0⟩.
        EC-01: Ĥ|0⟩ = 0 → Todas las partículas en fase.
        """
        # Vector de estado: superposición coherente
        self.vector_estado = np.ones(self.n, dtype=complex) / np.sqrt(self.n)
        
        # Hamiltoniano del Silencio (operador de acoplamiento)
        for i in range(self.n):
            self.hamiltoniano[i, i] = -1.0
            for j in self.particulas[i].vecinos:
                if j < self.n:
                    self.hamiltoniano[i, j] = 1.0 / len(self.particulas[i].vecinos)
        
        # Verificar EC-01: Ĥ|0⟩ = E₀|0⟩ con E₀ ≈ 0
        H_psi = self.hamiltoniano @ self.vector_estado
        self.E0_actual = np.abs(np.vdot(self.vector_estado, H_psi))
        
        print(f"✅ Estado fundamental inicializado")
        print(f"   E₀ = {self.E0_actual:.6e} (target: {self.energia_vacio:.6e})")
    
    def calcular_observables(self) -> Dict:
        """
        Calcula todos los observables cuánticos.
        EC-01 a EC-04.
        """
        # EC-01: Energía del vacío
        H_psi = self.hamiltoniano @ self.vector_estado
        E0 = np.real(np.vdot(self.vector_estado, H_psi))
        
        # EC-02: Norma del estado (debe ser 1)
        norma = np.real(np.vdot(self.vector_estado, self.vector_estado))
        
        # EC-03: Conmutador [Ĥ, P̂] (operador de traslación)
        # En el Silencio, el conmutador debe anularse
        P = np.eye(self.n) - np.roll(np.eye(self.n), 1, axis=0)
        conmutador = self.hamiltoniano @ P - P @ self.hamiltoniano
        norma_conmutador = np.linalg.norm(conmutador @ self.vector_estado)
        
        # EC-04: Dimensión del kernel de Ĥ
        autovalores = np.linalg.eigvalsh(self.hamiltoniano)
        dim_ker = np.sum(np.abs(autovalores) < 1e-10)
        
        # Fase global (debe ser cero en el Silencio)
        fase_global = np.angle(np.sum(self.vector_estado))
        
        # Coherencia cuántica
        matriz_densidad = np.outer(self.vector_estado, np.conj(self.vector_estado))
        coherencia = np.sum(np.abs(matriz_densidad)) - np.trace(np.abs(matriz_densidad))
        
        return {
            'E0': E0,
            'norma': norma,
            'conmutador': norma_conmutador,
            'dim_ker': dim_ker,
            'fase_global': fase_global,
            'coherencia': coherencia,
            'tiempo': self.tiempo_actual
        }
    
    def perturbar(self, intensidad: float = 0.001) -> None:
        """
        Aplica una perturbación cuántica para Montecarlo.
        Simula fluctuaciones del vacío.
        """
        perturbacion = np.random.normal(0, intensidad, self.n) + \
                       1j * np.random.normal(0, intensidad, self.n)
        
        # Proyectar ortogonal al estado actual
        psi = self.vector_estado
        perturbacion = perturbacion - np.vdot(psi, perturbacion) * psi
        
        # Normalizar
        norma = np.sqrt(np.vdot(perturbacion, perturbacion))
        if norma > 1e-15:
            perturbacion = perturbacion / norma
        
        # Aplicar perturbación
        angulo = intensidad * np.random.random()
        self.vector_estado = np.cos(angulo) * psi + np.sin(angulo) * perturbacion
        self.vector_estado = self.vector_estado / np.sqrt(np.vdot(self.vector_estado, self.vector_estado))

        """
        Evoluciona el Silencio un paso temporal.
        """
        if dt is None:
            dt = self.config['dt']
        
        self.tiempo_actual += dt
        
        # Evolución unitaria
        U = np.eye(self.n, dtype=complex) - 1j * dt * self.hamiltoniano
        self.vector_estado = U @ self.vector_estado
        self.vector_estado = self.vector_estado / np.sqrt(
            np.vdot(self.vector_estado, self.vector_estado)
        )
        
        # Ruido cuántico (fluctuaciones del vacío)
        self.perturbar(1e-4)
        
        return self.calcular_observables()
    
    def detectar_ruptura(self) -> Optional[int]:
        """
        Detecta la primera ruptura del Silencio.
        Ocurre cuando una partícula se desfasa significativamente.
        """
        fases = np.angle(self.vector_estado)
        fases_abs = np.abs(fases)
        
        # Umbral de ruptura
        umbral = self.config['temperatura_ruptura']
        
        particulas_rotas = np.where(fases_abs > umbral)[0]
        
        if len(particulas_rotas) > 0:
            return particulas_rotas[np.argmax(fases_abs[particulas_rotas])]
        return None
    
    def calcular_deuda_informacion(self) -> float:
        """
        Calcula la Deuda de Información 𝔇 del sistema.
        EC-10: 𝔇 = ABC - 2abc
        """
        # Mapear a parámetros de bifurcación
        fases = np.angle(self.vector_estado)
        
        A = 1.0 + np.std(np.real(self.vector_estado))
        B = 1.0 + np.std(np.imag(self.vector_estado))
        a = 0.5 + np.mean(np.abs(fases[:self.n//2]))
        b = 0.5 + np.mean(np.abs(fases[self.n//2:]))
        c = 1.0
        C = A / (a + 1e-10)
        
        D = A * B * C - 2 * a * b * c
        return D

        return stats_dict

class PrimeraRuptura:
    """
    Simula la primera autocomparación que rompe el Silencio.
    El momento exacto donde emerge la estructura.
    """
    
    def __init__(self, silencio: SilencioArmonico):
        self.silencio = silencio
        self.tiempo_ruptura = None
        self.particula_disparadora = None
        self.historia_fases = []
        self.historia_energia = []
        self.historia_deuda = []
        
    def simular(self, tiempo_max: float = 5.0, dt: float = 0.01) -> RupturaSilencio:
        """
        Simula la evolución hasta la primera ruptura.
        """
        print(f"\n💥 SIMULANDO PRIMERA RUPTURA")
        print(f"   Tiempo máximo: {tiempo_max}s")
        print(f"   Paso: {dt}s")
        
        t = 0.0
        paso = 0
        
        while t < tiempo_max:
            # Evolucionar
            obs = self.silencio.evolucionar(dt)
            
            # Registrar
            fases = np.angle(self.silencio.vector_estado)
            self.historia_fases.append(np.abs(fases))
            self.historia_energia.append(obs['E0'])
            self.historia_deuda.append(self.silencio.calcular_deuda_informacion())
            
            # Detectar ruptura
            particula_rota = self.silencio.detectar_ruptura()
            
            if particula_rota is not None and self.tiempo_ruptura is None:
                self.tiempo_ruptura = t
                self.particula_disparadora = particula_rota
                
                print(f"\n   ⚡ ¡RUPTURA DETECTADA! ⚡")
                print(f"   Tiempo: t = {t:.4f}s")
                print(f"   Partícula disparadora: #{particula_rota}")
                print(f"   Fase: {fases[particula_rota]:.4f} rad")
                print(f"   Energía: {obs['E0']:.6e}")
                
                # Crear patrón de bifurcación
                patron = self._extraer_patron_bifurcacion()
                
                ruptura = RupturaSilencio(
                    tiempo_ruptura=t,
                    particula_disparadora=particula_rota,
                    patron_bifurcacion=patron,
                    deuda_informacion=self.silencio.calcular_deuda_informacion(),
                    energia_liberada=obs['E0']
                )
                
                self.silencio.ruptura = ruptura
                return ruptura
            
            t += dt
            paso += 1
        
        print(f"\n   ⚠️ No se detectó ruptura en {tiempo_max}s")
        return None
    
    def _extraer_patron_bifurcacion(self) -> np.ndarray:
        """
        Extrae el patrón espacial de la bifurcación.
        Muestra cómo emerge A=1, B=1, a=0.5, b=0.5.
        """
        fases = np.angle(self.silencio.vector_estado)
        n = len(fases)
        
        # Crear matriz de bifurcación
        patron = np.zeros((n, 4))
        
        for i in range(n):
            p = self.silencio.particulas[i]
            
            # A y B: amplitudes
            patron[i, 0] = np.abs(np.real(self.silencio.vector_estado[i]))  # A
            patron[i, 1] = np.abs(np.imag(self.silencio.vector_estado[i]))  # B
            
            # a y b: fases (sombras de Fresnel)
            patron[i, 2] = np.abs(np.cos(fases[i]))  # a
            patron[i, 3] = np.abs(np.sin(fases[i]))  # b
        
        return patron

class VisualizadorSilencio:

    def visualizar_red_silencio(self, ax: plt.Axes = None):
        """Visualiza la red hexagonal del Silencio en equilibrio perfecto"""

            for j in p.vecinos:
                if j < len(self.silencio.particulas):
                    q = self.silencio.particulas[j]
                    ax.plot([p.x, q.x], [p.y, q.y], 'white', alpha=0.08, linewidth=0.5)
        
        # Dibujar partículas
        for p in self.silencio.particulas:

            circle = Circle((p.x, p.y), 0.15, facecolor=color, edgecolor='white',

        ax.axis('off')
        
        return ax
    
    def visualizar_ruptura(self, ruptura: RupturaSilencio, ax: plt.Axes = None):
        """Visualiza el momento exacto de la primera ruptura"""
        if ax is None:
            fig, ax = plt.subplots(1, 1, figsize=(10, 10))
        
        ax.set_facecolor('#0a0a0a')
        
        fases = np.angle(self.silencio.vector_estado)
        max_fase = np.max(np.abs(fases))
        
        # Dibujar conexiones
        for p in self.silencio.particulas:
            for j in p.vecinos:
                if j < len(self.silencio.particulas):
                    q = self.silencio.particulas[j]
                    ax.plot([p.x, q.x], [p.y, q.y], 'white', alpha=0.05, linewidth=0.5)
        
        # Dibujar partículas con color según fase
        for i, p in enumerate(self.silencio.particulas):
            fase_norm = np.abs(fases[i]) / (max_fase + 1e-10)
            
            if i == ruptura.particula_disparadora:
                color = '#ff4444'  # Rojo brillante: partícula disparadora
                tamano = 0.4
                edgecolor = 'yellow'
                linewidth = 3
            else:
                color = plt.cm.RdYlBu_r(fase_norm)
                tamano = 0.12 + 0.15 * fase_norm
                edgecolor = 'white'
                linewidth = 0.5
            
            circle = Circle((p.x, p.y), tamano, facecolor=color, 
                          edgecolor=edgecolor, linewidth=linewidth, alpha=0.9)
            ax.add_patch(circle)
        
        # Anotación de la ruptura
        p_rota = self.silencio.particulas[ruptura.particula_disparadora]
        ax.annotate(f'⚡ RUPTURA\nt={ruptura.tiempo_ruptura:.3f}s', 
                   xy=(p_rota.x, p_rota.y),
                   xytext=(p_rota.x + 2, p_rota.y + 2),
                   color='yellow', fontsize=10, fontweight='bold',
                   arrowprops=dict(arrowstyle='->', color='yellow', lw=2))
        
        ax.set_xlim(-self.silencio.radio, self.silencio.radio)
        ax.set_ylim(-self.silencio.radio, self.silencio.radio)
        ax.set_aspect('equal')
        ax.set_title(f'PRIMERA AUTORRUPTURA DEL SILENCIO\nt = {ruptura.tiempo_ruptura:.4f}s | Partícula #{ruptura.particula_disparadora}', 
                    color='white', fontsize=14, fontweight='bold')
        ax.axis('off')
        
        return ax
    
    def visualizar_patron_bifurcacion(self, ruptura: RupturaSilencio):
        """Visualiza el patrón de bifurcación A, B, a, b"""
        fig, axes = plt.subplots(2, 2, figsize=(12, 10))
        fig.patch.set_facecolor('#0a0a0a')
        
        patron = ruptura.patron_bifurcacion
        titulos = ['A (Amplitud Real)', 'B (Amplitud Imaginaria)', 
                   'a (Sombra cos)', 'b (Sombra sin)']
        cmaps = ['Reds', 'Blues', 'Greens', 'Oranges']
        
        for i, ax in enumerate(axes.flat):
            ax.set_facecolor('#0a0a0a')
            
            # Reconstruir coordenadas
            valores = patron[:, i]
            
            scatter = ax.scatter(
                [p.x for p in self.silencio.particulas],
                [p.y for p in self.silencio.particulas],
                c=valores, cmap=cmaps[i], s=100, edgecolors='white', linewidth=0.5
            )
            
            plt.colorbar(scatter, ax=ax, label=titulos[i])
            ax.set_title(titulos[i], color='white', fontsize=12)
            ax.set_aspect('equal')
            ax.axis('off')
        
        plt.suptitle('PATRÓN DE BIFURCACIÓN: EMERGE LA ESTRUCTURA\nA=1, B=1, a=0.5, b=0.5', 
                    color='white', fontsize=14, fontweight='bold')
        plt.tight_layout()
        return fig

    def animar_ruptura(self, historia_fases: List, historia_deuda: List, 
                      tiempo_ruptura: float, archivo_salida: str = 'silencio_ruptura.gif'):
        """Crea una animación de la ruptura del Silencio"""
        print(f"\n🎬 Creando animación...")
        
        fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(16, 8))
        fig.patch.set_facecolor('#0a0a0a')
        
        n_frames = min(len(historia_fases), 200)
        
        def actualizar(frame):
            ax1.clear()
            ax2.clear()
            ax1.set_facecolor('#0a0a0a')
            ax2.set_facecolor('#0a0a0a')
            
            # Panel izquierdo: Red con fases
            fases = historia_fases[frame]
            max_fase = np.max(fases) if np.max(fases) > 0 else 1.0
            
            for p in self.silencio.particulas:
                for j in p.vecinos:
                    if j < len(self.silencio.particulas):
                        q = self.silencio.particulas[j]
                        ax1.plot([p.x, q.x], [p.y, q.y], 'white', alpha=0.05, linewidth=0.5)
            
            for i, p in enumerate(self.silencio.particulas):
                fase_norm = fases[i] / max_fase
                color = plt.cm.RdYlBu_r(fase_norm) if fase_norm > 0.01 else '#444444'
                tamano = 0.12 + 0.3 * fase_norm
                
                circle = Circle((p.x, p.y), tamano, facecolor=color, 
                              edgecolor='white', linewidth=0.5, alpha=0.9)
                ax1.add_patch(circle)
            
            t_actual = frame * 0.05
            ax1.set_xlim(-self.silencio.radio, self.silencio.radio)
            ax1.set_ylim(-self.silencio.radio, self.silencio.radio)
            ax1.set_aspect('equal')
            ax1.set_title(f'SILENCIO ARMÓNICO\nt = {t_actual:.2f}s', color='white', fontsize=12)
            ax1.axis('off')
            
            # Panel derecho: Deuda de Información vs tiempo
            t_vals = np.linspace(0, t_actual, frame + 1)
            deuda_vals = historia_deuda[:frame + 1]
            
            ax2.plot(t_vals, deuda_vals, 'gold', linewidth=2)
            ax2.axhline(1.5, color='red', linestyle='--', linewidth=1, label='𝔇=1.5 (Bifurcación)')
            ax2.axvline(tiempo_ruptura, color='cyan', linestyle='--', linewidth=2, 
                       label=f'Ruptura: t={tiempo_ruptura:.3f}s' if tiempo_ruptura else '')
            ax2.set_xlabel('Tiempo (s)', color='white')
            ax2.set_ylabel('Deuda 𝔇', color='white')
            ax2.set_title('DEUDA DE INFORMACIÓN', color='white', fontsize=12)
            ax2.legend()
            ax2.tick_params(colors='white')
            ax2.set_ylim(0, 3)
        
        anim = FuncAnimation(fig, actualizar, frames=n_frames, interval=50)
        anim.save(archivo_salida, writer=PillowWriter(fps=20))
        print(f"✅ Animación guardada: {archivo_salida}")
        
        return anim
